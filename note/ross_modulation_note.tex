\title{Speeding up a Markov environment can increase mean workload:\\
a counterexample to the universal form of Ross's second conjecture}
\author{Tynan Daly}
\date{10 September 2026}

\begin{document}
\maketitle

\begin{abstract}
Consider a single-server queue with Cox arrivals of intensity $\lambda_c(t)=\lambda(ct)$, where $\lambda$ is a stationary ergodic process and $c>0$ is the modulation speed. Ross's second conjecture (1978) asserts that the mean stationary workload $w(c)$ is nonincreasing in $c$; Leskel\"a (2022) records it as open in general and poses it as Problem~1. We give an explicit 64-state irreducible Markov intensity process, with strictly positive transition rates between every pair of states and unit deterministic service, for which $w(3\pi)>w(2\pi)$. The proof is a two-sided bound on $w(c)$ in terms of a single autocorrelation integral $K(c)$, and shows more generally that $K$ nonincreasing in $c$ is necessary for the conjecture to hold at all light loads. The mechanism is resonance between the environment's oscillation and the service time.
\end{abstract}

\section{Introduction}

Let $\lambda=(\lambda(t))_{t\in\mathbb R}$ be a stationary ergodic nonnegative process and, conditionally on $\lambda$, let arrivals to a single server form a Poisson process of intensity $\lambda_c(t)=\lambda(ct)$. Service requirements are i.i.d.\ and independent of the arrivals, and the system is stable. Write $w(c)$ for the mean stationary workload. Ross~\cite{ross} conjectured two things about this system. The first, that $w(c)$ is at least the workload of the M/G/1 queue with constant rate $\E\lambda(0)$, was proved by Rolski~\cite{rolski81}. The second is that a faster environment averages out burstiness and so
\begin{equation}
c_1\le c_2\ \Longrightarrow\ w(c_1)\ge w(c_2).
\label{eq:conj}
\end{equation}
This has been proved for finite-state Markov environments with additional order structure: Ross~\cite{ross} for a two-state chain with one zero state, Chang, Chao and Pinedo~\cite{ccp} for chains with $q_{ij}=\alpha_i$ for all $j\ne i$, and B\"auerle and Rolski~\cite{br} when the generator and its time reversal are both stochastically monotone. Leskel\"a~\cite{leskela} surveys these results and asks, as Problem~1, whether \eqref{eq:conj} holds for every stationary ergodic $\lambda$ and, if not, for a characterization.

We answer the first question in the negative inside the class of finite-state Markov environments.

\begin{theorem}\label{thm:main}
Let $n=64$, $\theta=2\pi/n$, and let $J$ be the continuous-time Markov chain on $\{0,\dots,n-1\}$ with generator
\begin{equation}
Q=r(S-I)+\delta(\Pi-I),\qquad r=1/\sin\theta,\quad \delta=1/2000,
\label{eq:generator}
\end{equation}
where $(Sv)_j=v_{j+1\bmod n}$ and $\Pi$ has every entry $1/n$. Every off-diagonal entry of $Q$ is strictly positive, and the uniform distribution is stationary. Set
\[
x_j=1+\tfrac12\cos\!\big(\theta(j+\tfrac14)\big),\qquad \lambda(t)=\eps\,x_{J(t)},\qquad \eps=1/2000,
\]
and let every service requirement equal one. Then, with $w(c)$ the mean stationary workload under intensity $\lambda(ct)$,
\[
w(3\pi)-w(2\pi)>\frac{1587}{25\,580\,800\,000\,000}>0 .
\]
\end{theorem}

The $x_j$ are distinct, so $\lambda$ is itself a Markov chain, not a function of a hidden one; it is irreducible and mixing with $\E x_{J(0)}=1$ and $\tfrac12<x_j<\tfrac32$. The chain is not stochastically monotone in the intensity ordering (Section~\ref{sec:markov}), so Theorem~\ref{thm:main} does not contradict~\cite{br} or~\cite{ccp}. Rolski's lower bound~\cite{rolski81} is likewise untouched.

Section~\ref{sec:bound} proves a two-sided bound on $w(c)$ valid for any bounded stationary intensity; Section~\ref{sec:markov} evaluates it for the chain above; Section~\ref{sec:mechanism} explains the mechanism and derives a necessary condition for~\eqref{eq:conj}; Section~\ref{sec:scope} states what is and is not claimed.

\section{A two-sided bound on the mean workload}\label{sec:bound}

Let $X$ be stationary with $\E X(0)=1$ and $0\le X\le M$, let arrivals be Cox with intensity $\eps X(ct)$, let every service requirement be~$1$, and assume $\eps M<1$. Define
\begin{equation}
K(c)=\int_0^1(1-u)\,\E[X(0)X(cu)]\,du.
\label{eq:K}
\end{equation}

\begin{lemma}\label{lem:bound}
The stationary mean workload exists and satisfies
\begin{equation}
\frac{\eps}{2}+\eps^2K(c)\ \le\ w(c)\ \le\ \frac{\eps}{2}+\frac{\eps^2K(c)}{1-\eps M}.
\label{eq:bound}
\end{equation}
Consequently
\begin{equation}
w(c_2)-w(c_1)\ \ge\ \eps^2\Big[K(c_2)-\frac{K(c_1)}{1-\eps M}\Big].
\label{eq:comparison}
\end{equation}
\end{lemma}

\begin{proof}
\emph{Existence.} Realize the arrivals by thinning a homogeneous Poisson process of rate $\eps M$, independent of $X$, keeping a point at $t$ with probability $X(ct)/M$. Start the queue empty at time $-T$. The workload at time~$0$ is nondecreasing in $T$ and dominated by the stationary workload of the M/D/1 queue fed by all proposal points, which has load $\eps M<1$ and finite moments of every order. The limit exists, is jointly stationary with $X$ by shift-equivariance, and has finite second moment.

\emph{Balance.} Write $V$ for the stationary workload and $N$ for the arrival process. Pathwise,
\[
V(t)^2-V(0)^2=-2\int_0^t V(s)\,ds+\int_{(0,t]}\big(2V(s-)+1\big)N(ds).
\]
With respect to a filtration containing the whole path of $X$ and the arrival history, $N$ has intensity $\eps X(cs)$. Taking expectations, using stationarity and the moment bounds,
\begin{equation}
w(c)=\frac{\eps}{2}+\eps\,\E[X(0)V(0)].
\label{eq:balance}
\end{equation}

\emph{Comparison.} Let $Y=\sum_{T_i\in(-1,0]}(1+T_i)$ be the residual work if every job were served on its own unit-speed server. A job of age $u<1$ has received at most $u$ units of service from a single work-conserving server, so $0\le Y\le V(0)$. Conditioning on $X$,
\[
\E Y=\frac{\eps}{2},\qquad \E[X(0)Y]=\eps\int_0^1(1-u)\E[X(0)X(-cu)]\,du=\eps K(c),
\]
using $\E[X(0)X(-t)]=\E[X(0)X(t)]$ by stationarity. Put $D=V(0)-Y\ge0$ and $d=\E D=w(c)-\eps/2$. Substituting into~\eqref{eq:balance},
\[
d=\eps^2K(c)+\eps\,\E[X(0)D].
\]
Since $0\le X\le M$ and $D\ge0$, we have $0\le\E[X(0)D]\le M d$. The lower bound in~\eqref{eq:bound} follows from the left inequality; solving $d\le\eps^2K(c)+\eps M d$ gives the upper bound. Subtracting gives~\eqref{eq:comparison}.
\end{proof}

The upper bound controls the entire remainder above second order, not one term of an expansion; this is what makes the strict inequality in Theorem~\ref{thm:main} a finite-load statement.

\section{The Markov environment}\label{sec:markov}

\paragraph{Autocorrelation.} On the first Fourier mode $v_j=e^{i\theta j}$, $\Pi$ acts as zero and $S$ as multiplication by $e^{i\theta}$, so
\[
Qv=(-\alpha+i)\,v,\qquad \alpha=\tan(\pi/64)+\delta .
\]
Averaging over the uniform stationary law,
\begin{equation}
\E[X(0)X(t)]=1+\tfrac18\,e^{-\alpha t}\cos t,\qquad t\ge0,
\label{eq:autocorr}
\end{equation}
and therefore, with $z=(-\alpha+i)c$,
\begin{equation}
K(c)=\frac12+\frac18\,I_\alpha(c),\qquad I_\alpha(c)=\operatorname{Re}\frac{e^{z}-1-z}{z^2}.
\label{eq:kernel}
\end{equation}

\paragraph{Certificate.} The following uses only elementary inequalities and rational arithmetic. From $3<\pi<22/7$, $\sin x<x$ and $\cos x>1-x^2/2$ with $u=11/224$,
\[
0<\alpha<\frac{u}{1-u^2/2}+\frac1{2000}=\frac{9956231}{200462000}<\frac1{20}.
\]
Let $A=\alpha/(1+\alpha^2)$ and $C=(1-\alpha^2)/(1+\alpha^2)^2$, so $A<\alpha$ and $99/100<C<1$. At multiples of $\pi$,
\[
I_\alpha(k\pi)=\frac{A}{k\pi}+C\,\frac{1-(-1)^ke^{-\alpha k\pi}}{(k\pi)^2}.
\]
Using $e^{-x}\ge1-x$ and $1-e^{-x}\le x$,
\[
I_\alpha(3\pi)-I_\alpha(2\pi)>-\frac1{360}+\frac{99}{100}\Big[\frac{2}{9(22/7)^2}-\frac1{72}\Big]=\frac{91}{15840}>\frac1{200},
\]
the bracket being positive. Hence $K(3\pi)-K(2\pi)>1/1600$. Also $I_\alpha(2\pi)\le\alpha/\pi<1/60$, so $K(2\pi)<241/480$. With $M=3/2$, \eqref{eq:comparison} gives
\[
w(3\pi)-w(2\pi)>\frac1{2000^2}\Big[\frac1{1600}-\frac{(3/4000)(241/480)}{1-3/4000}\Big]=\frac{1587}{25\,580\,800\,000\,000},
\]
which proves Theorem~\ref{thm:main}. (Numerically $K(2\pi)\approx0.5018268$, $K(3\pi)\approx0.5029285$, and the true gap is about $1.8\times10^{-10}$; these decimals play no role in the proof.)

\paragraph{Outside the known positive results.} Order the states by intensity: $x_{62}<x_1<x_{63}<x_0$. Let $H=\{x_{63},x_0\}$. From state~$62$ the forward move enters $H$ at rate~$r$; from state~$1$ it does not, and the redraw contributes equally from both. So $P_{62}(X(t)\in H)-P_1(X(t)\in H)=rt+O(t^2)>0$ for small $t$, although $x_{62}<x_1$: the chain is not stochastically monotone in the intensity order, so~\cite{br} does not apply. It has no uniform-rate structure, so~\cite{ccp} does not apply either.

\paragraph{Robustness.} The certificate is strict, and $K(c)$, the stationary law, and the stability margin depend continuously on the generator and the intensity levels. Hence the reversal persists in an open neighborhood of the example (renormalizing intensities to mean one if desired). We do not estimate the size of that neighborhood.

\begin{remark}[A 24-state rational example]
A directed 24-cycle with unit forward rates, rational intensity levels, $\eps=10^{-6}$ and speeds $c_1=29$, $c_2=30$ also reverses, with a certified gap exceeding $10^{-18}$. The certificate encloses $K$ by exact rational matrix-series bounds; see \texttt{verify/verify\_markov\_24.py} in the repository. It trades effect size for fewer states and rational data. Neither state count is claimed minimal.
\end{remark}

\section{Mechanism, and a necessary condition}\label{sec:mechanism}

\begin{remark}[The periodic case]\label{rem:periodic}
Take $X(t)=1+\cos(t+\Theta)$ with $\Theta$ uniform on $[0,2\pi)$, so $M=2$. This is stationary and ergodic, though not mixing or Markov. Then $\E[X(0)X(t)]=1+\tfrac12\cos t$ and
\[
K(c)=\frac12+\frac{1-\cos c}{2c^2}.
\]
At $c=2\pi$ the correlation term vanishes identically: the environment completes exactly one period per unit service time, so the intensity seen over any service window integrates to its mean. At $c=3\pi$ it does not, and $K(3\pi)=\tfrac12+1/(9\pi^2)>K(2\pi)=\tfrac12$. Lemma~\ref{lem:bound} then gives $w(3\pi)>w(2\pi)$ for every mean rate $\rho<1/92$; at $\rho=1/100$ the certified gap is $1/11\,025\,000$. The Markov chain of Theorem~\ref{thm:main} is a damped version of this environment: exponential holding times replace $\cos t$ by $e^{-\alpha t}\cos t$ in~\eqref{eq:autocorr}, which weakens the resonance without destroying it.
\end{remark}

The bound~\eqref{eq:bound} also says something about the second half of Leskel\"a's Problem~1.

\begin{proposition}\label{prop:necessary}
Let $X$ be stationary with $\E X(0)=1$, $0\le X\le M$, and unit deterministic service. Then $w(c)=\eps/2+\eps^2K(c)+O(\eps^3)$ uniformly in $c$ as $\eps\downarrow0$. In particular, if \eqref{eq:conj} holds for all sufficiently small $\eps$, then $K$ is nonincreasing on $(0,\infty)$.
\end{proposition}

\begin{proof}
Since $\E[X(0)X(t)]\le M\,\E X(t)=M$, we have $K(c)\le M/2$ for all $c$, so by~\eqref{eq:bound}
\[
0\ \le\ w(c)-\frac{\eps}{2}-\eps^2K(c)\ \le\ \frac{\eps^3 M K(c)}{1-\eps M}\ \le\ \frac{\eps^3M^2}{2(1-\eps M)},
\]
uniformly in $c$. If $c_1<c_2$ and $K(c_2)>K(c_1)$, then~\eqref{eq:comparison} gives $w(c_2)>w(c_1)$ as soon as $\eps M K(c_1)<(1-\eps M)\big(K(c_2)-K(c_1)\big)$, which holds for all small $\eps$, contradicting~\eqref{eq:conj}.
\end{proof}

Since $K(c)=\int_0^1(1-u)R(cu)\,du$ with $R$ the autocorrelation of $X$, this is a condition on $R$ alone. It holds when $R$ is nonincreasing---in particular for reversible stochastically monotone chains, consistent with~\cite{br}---and fails whenever $R$ oscillates with a period commensurate with the service time, as in both examples above.

\section{Scope}\label{sec:scope}

Theorem~\ref{thm:main} shows that finite-state Markov modulation, even with all transition rates positive, does not by itself imply~\eqref{eq:conj}. It does not identify the class of environments for which~\eqref{eq:conj} holds; Proposition~\ref{prop:necessary} gives a necessary condition at light load only. The service law is deterministic, which Problem~1 permits; whether a reversal occurs with exponential service is open, and the resonance mechanism suggests it may not. The certified effect is small in absolute terms ($\sim10^{-10}$ at load $1/2000$), of relative order $\eps\,\Delta K\approx10^{-6}$; the question is qualitative.

We have not established that this example is new. The periodic-input literature of the late 1980s~\cite{rolski87,rolski89,lemoine} contains light-traffic and exact analyses in which the integral~\eqref{eq:K} or an equivalent may appear, and~\cite{heyman,mr} discuss Ross's conjectures directly; we have not had full access to these papers.

\paragraph{Reproducibility.} Exact-arithmetic verifiers for the certificates in Theorem~\ref{thm:main}, the 24-state example, and Remark~\ref{rem:periodic} (Python~3.10, standard library) are archived with this note at Zenodo, \href{https://doi.org/10.5281/zenodo.22686842}{doi:10.5281/zenodo.22686842}.
